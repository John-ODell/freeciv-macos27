# Citizens, specialists and the top bar's indicators in freeciv45iso (issue 298). Run on a driver build at patch 0057:
#   FREECIV_DRIVE_PREFIX=<prefix> FREECIV_DRIVE_TILESET=freeciv45iso FREECIV_DRIVE_TILESETS=<repo>/tilesets \
#     scripts/ui-drive.sh notes/freeciv45/ui/citizens-test/citizens.txt <outdir>
# Lua founds Manyhands on the free land tile near the start with the most land around it and has three Settlers join
# it (size 7, so some citizens are unhappy and the city takes entertainers). Then the player gets Railroad, Automobile
# and Rocketry in turn, the techs civ45's styles.ruleset ties to the industrial, electric age and modern styles, with a
# turn between so the city's style is refreshed; each era gets a shot of the city dialog. Lua output goes to the chat.
pace 1200
wait 4000
dismiss
wait 1000
option popup_new_cities off
chat /lua cmd fc45look = {}; for p in players_iterate() do if p:is_human() then fc45look.me = p end end; for x in fc45look.me:units_iterate() do fc45look.start = x.tile; break end; local best = -1; for t in fc45look.start:square_iterate(6) do if t.terrain:class_name() == "Land" and t:num_units() == 0 then local n = 0; for u in t:square_iterate(2) do if u.terrain:class_name() == "Land" then n = n + 1 end end; if n > best then best = n; fc45look.home = t end end end; edit.create_city(fc45look.me, fc45look.home, "Manyhands")
wait 2500
chat /lua cmd local c = fc45look.home:city(); for i = 1, 3 do local u = edit.create_unit(fc45look.me, fc45look.home, find.unit_type("Settlers"), 0, nil, -1); log.normal("FC45LOOK join %s", tostring(edit.perform_action(u, find.action("Join City"), c))) end; log.normal("FC45LOOK size %d", c:size())
wait 2500
dismiss
opencity Manyhands
wait 2500
state ancient
shot ancient
key Escape
wait 1000
chat /lua cmd edit.give_tech(fc45look.me, find.tech_type("Railroad"), -1, false, "test")
wait 1500
endturn
wait 9000
dismiss
opencity Manyhands
wait 2500
state industrial
shot industrial
key Escape
wait 1000
chat /lua cmd edit.give_tech(fc45look.me, find.tech_type("Automobile"), -1, false, "test")
wait 1500
endturn
wait 9000
dismiss
opencity Manyhands
wait 2500
state electricage
shot electricage
key Escape
wait 1000
chat /lua cmd edit.give_tech(fc45look.me, find.tech_type("Rocketry"), -1, false, "test")
wait 1500
endturn
wait 9000
dismiss
opencity Manyhands
wait 2500
state modern
shot modern
quit
