pace 1200
wait 4000
dismiss
wait 1000
state start
chat /lua cmd notify.event(find.unit(nil, $UNIT).owner, find.tile($X+3, $Y), E.UNIT_WIN_DEF, "Your Warriors survived the attack of a Barbarian Archer.")
wait 800
chat /lua cmd notify.event(find.unit(nil, $UNIT).owner, find.tile($X-3, $Y+1), E.MY_DIPLOMAT_EMBASSY, "Your Explorer met the envoys of the Babylonians.")
wait 800
chat /lua cmd notify.event(find.unit(nil, $UNIT).owner, find.tile($X, $Y+3), E.HUT_GOLD, "You found 25 gold in a hut.")
wait 800
chat /lua cmd notify.event(find.unit(nil, $UNIT).owner, find.tile($X+1, $Y-2), E.TECH_GAIN, "Learned Bronze Working from the ruins.")
wait 800
chat /lua cmd edit.create_city(find.unit(nil, $UNIT).owner, find.tile($X, $Y), "Testburg")
wait 2500
dismiss
wait 500
click button Return to map
wait 1000
chat /lua cmd notify.event(find.unit(nil, $UNIT).owner, find.tile($X, $Y), E.CITY_GROWTH, "Testburg grows to size 2.")
wait 1500
state five-messages
shot 01-badges
hover tip Barbarian Archer
wait 1500
shot 02-hover-military
click tip Barbarian Archer
wait 1500
state after-military
shot 03-military-visited
click tip Testburg
wait 2000
state after-city
shot 04-city-badge-overlay
click button Return to map
wait 1500
state overlay-closed
chat /lua cmd for i = 1, 10 do notify.event(find.unit(nil, $UNIT).owner, find.tile($X+(i%5), $Y+2), E.UNIT_LOST_MISC, "Test message %d: a unit was lost.", i) end
wait 2500
state many
shot 05-plus-badge
hover tip more messages
wait 1500
shot 06-hover-plus
click tip more messages
wait 1500
state plus-clicked
shot 07-plus-log
option gui_qt_freeciv45_edges off
wait 1000
state edges-off
option gui_qt_freeciv45_edges on
wait 1000
state edges-on
endturn

dismiss
wait 1000
state turn2
shot 08-turn2
