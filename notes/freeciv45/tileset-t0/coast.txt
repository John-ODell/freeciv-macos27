# issue 184: south of the start, deep ocean on the west and ocean on the east, with a one-tile grassland island in
# the deep ocean and a 2 x 2 lake to the east, so every water depth meets another and the land
pace 1000
wait 4000
dismiss
wait 1000
chat /lua cmd local function pick(dx, dy) if dx == -6 and dy == 4 then return "Grassland" end; if dy >= 2 and dy <= 6 and dx >= -10 and dx <= -1 then if dx <= -5 then return "Deep Ocean" end; return "Ocean" end; if dy >= 3 and dy <= 4 and dx >= 2 and dx <= 3 then return "Lake" end; return nil end; for p in players_iterate() do if p:is_human() then for u in p:units_iterate() do local x0, y0 = u.tile.nat_x, u.tile.nat_y; for dy = -6, 7 do for dx = -10, 10 do local t = find.tile(x0 + dx, y0 + dy); local n = pick(dx, dy); if t and n then edit.change_terrain(t, find.terrain(n)) end end end; break end end end
wait 15000
dismiss
shot coast
quit
